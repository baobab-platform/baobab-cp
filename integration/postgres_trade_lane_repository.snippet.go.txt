func (r *PostgresRepository) GetTradeLane(
	ctx context.Context,
	tenantID string,
	tradeLaneID string,
) (domain.TradeLane, error) {
	const query = `
		SELECT trade_lane_id, tenant_id, origin_market_id, destination_market_id,
		       direction, status, permitted_capability_keys, created_at, updated_at
		  FROM market.trade_lane
		 WHERE tenant_id = $1
		   AND trade_lane_id = $2`

	var lane domain.TradeLane
	err := r.db.QueryRow(ctx, query, tenantID, tradeLaneID).Scan(
		&lane.ID,
		&lane.TenantID,
		&lane.OriginMarketID,
		&lane.DestinationMarketID,
		&lane.Direction,
		&lane.Status,
		&lane.PermittedCapabilityKeys,
		&lane.CreatedAt,
		&lane.UpdatedAt,
	)
	if err != nil {
		return domain.TradeLane{}, fmt.Errorf("get trade lane: %w", err)
	}
	return lane, nil
}

func (r *PostgresRepository) SaveTradeLane(
	ctx context.Context,
	lane domain.TradeLane,
) error {
	if err := lane.Validate(); err != nil {
		return fmt.Errorf("validate trade lane: %w", err)
	}

	const query = `
		INSERT INTO market.trade_lane (
			trade_lane_id, tenant_id, origin_market_id, destination_market_id,
			direction, status, permitted_capability_keys, created_at, updated_at
		) VALUES ($1,$2,$3,$4,$5,$6,$7,$8,$9)
		ON CONFLICT (trade_lane_id) DO UPDATE SET
			direction = EXCLUDED.direction,
			status = EXCLUDED.status,
			permitted_capability_keys = EXCLUDED.permitted_capability_keys,
			updated_at = EXCLUDED.updated_at
		WHERE market.trade_lane.tenant_id = EXCLUDED.tenant_id`

	_, err := r.db.Exec(ctx, query,
		lane.ID,
		lane.TenantID,
		lane.OriginMarketID,
		lane.DestinationMarketID,
		lane.Direction,
		lane.Status,
		lane.PermittedCapabilityKeys,
		lane.CreatedAt,
		lane.UpdatedAt,
	)
	if err != nil {
		return fmt.Errorf("save trade lane: %w", err)
	}
	return nil
}
